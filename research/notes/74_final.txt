# Pico VR service. Stock definition from the 8.1 system image, plus a seclabel.
# Android 10 init refuses to start a service with no SELinux domain -
#   "init: service pvrservice does not have a SELinux domain defined"
# Proper fix for a real build: ship a pvrservice domain in sepolicy rather than
# the events. Without it pvrservice loads, then dies with
service pvrservice /system/bin/pvrservice
     seclabel u:r:shell:s0
     setenv LD_PRELOAD /system/lib64/libshim_pvr.so
on property:sys.pvr.vrservice.state=0
    stop pvrservice
on property:sys.pvr.vrservice.state=1
    start pvrservice
on property:sys.pvr.vrservice.state=2
    restart pvrservice
